SYSMIN Windows Server PowerShell Listener (Deep XML Extractor)

Extracts TargetUserName, WorkstationName, IpAddress, LogonType, ServiceName, and ProcessName directly from Event XML.

Grant Event Log Read Access (Optional): net localgroup "Event Log Readers" "$env:USERNAME" /add
Listener Script (Port 8080 — Extracts User & Hostname XML Fields) 
# SYSMIN Logs - Deep Entity Extraction Windows Server HTTP Agent
$IsAdmin = ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
$Port = 8080

$ServerHostname = $env:COMPUTERNAME
$ServerIP = (Get-NetIPAddress -AddressFamily IPv4 -ErrorAction SilentlyContinue | Where-Object { $_.InterfaceAlias -notmatch 'Loopback' -and $_.IPAddress -notlike '169.*' } | Select-Object -First 1).IPAddress
if (-not $ServerIP) { $ServerIP = "127.0.0.1" }

$Listener = New-Object System.Net.HttpListener
if ($IsAdmin) {
    $Listener.Prefixes.Add("http://+:$Port/")
} else {
    $Listener.Prefixes.Add("http://localhost:$Port/")
    $Listener.Prefixes.Add("http://127.0.0.1:$Port/")
}

try {
    $Listener.Start()
} catch {
    $Listener = New-Object System.Net.HttpListener
    $Listener.Prefixes.Add("http://localhost:$Port/")
    $Listener.Start()
}

Write-Host "=====================================================" -ForegroundColor DarkCyan
Write-Host "SYSMIN Agent Active on $ServerHostname (${ServerIP}:$Port)" -ForegroundColor Green
Write-Host "Local URL : http://localhost:$Port/api/logs" -ForegroundColor Cyan
Write-Host "=====================================================" -ForegroundColor DarkCyan

$Channels = @(
  'Security',
  'System',
  'Application',
  'Microsoft-Windows-PowerShell/Operational',
  'Microsoft-Windows-Windows Defender/Operational',
  'Microsoft-Windows-TerminalServices-LocalSessionManager/Operational',
  'Microsoft-Windows-TaskScheduler/Operational',
  'Microsoft-Windows-GroupPolicy/Operational'
)

while ($Listener.IsListening) {
    $Context = $Listener.GetContext()
    $Request = $Context.Request
    $Response = $Context.Response

    $Response.Headers.Add("Access-Control-Allow-Origin", "*")
    $Response.Headers.Add("Access-Control-Allow-Methods", "GET, OPTIONS")
    $Response.Headers.Add("Access-Control-Allow-Headers", "*")
    $Response.Headers.Add("Access-Control-Allow-Private-Network", "true")

    if ($Request.HttpMethod -eq "OPTIONS") {
        $Response.StatusCode = 200
        $Response.Close()
        continue
    }

    if ($Request.Url.AbsolutePath -eq "/api/logs") {
        Write-Host "[$(Get-Date -Format 'HH:mm:ss')] Sync request received! Extracting logs & user/host details..." -ForegroundColor Yellow
        $AllEvents = New-Object System.Collections.Generic.List[object]

        foreach ($Ch in $Channels) {
            try {
                $RawEvents = Get-WinEvent -LogName $Ch -MaxEvents 25 -ErrorAction Stop
                foreach ($Ev in $RawEvents) {
                    $TimeIso = if ($Ev.TimeCreated) { $Ev.TimeCreated.ToString("yyyy-MM-ddTHH:mm:ss.fffzzz") } else { (Get-Date).ToString("o") }
                    $TimeDisp = if ($Ev.TimeCreated) { $Ev.TimeCreated.ToString("dd MMM yyyy, hh:mm:ss tt") } else { (Get-Date).ToString("dd MMM yyyy, hh:mm:ss tt") }
                    $MsgText = try { $Ev.Message } catch { $null }

                    $XmlMap = @{}
                    try {
                        $XmlDoc = [xml]$Ev.ToXml()
                        foreach ($DataNode in $XmlDoc.Event.EventData.Data) {
                            if ($DataNode.Name -and $DataNode.'#text') {
                                $XmlMap[$DataNode.Name] = $DataNode.'#text'
                            }
                        }
                    } catch {}

                    if (-not $MsgText) {
                        $Pairs = ($XmlMap.GetEnumerator() | ForEach-Object { "$($_.Key): $($_.Value)" }) -join " | "
                        $MsgText = if ($Pairs) { "Event ID $($Ev.Id) - $Pairs" } else { "Event ID $($Ev.Id) logged by $($Ev.ProviderName) in $($Ev.LogName)." }
                    }

                    $AllEvents.Add([PSCustomObject]@{
                        Id               = $Ev.Id
                        RecordId         = $Ev.RecordId
                        LogName          = $Ev.LogName
                        ProviderName     = $Ev.ProviderName
                        LevelDisplayName = if ($Ev.LevelDisplayName) { $Ev.LevelDisplayName } else { "Information" }
                        Level            = $Ev.Level
                        TimeCreated      = $TimeIso
                        TimeDisplay      = $TimeDisp
                        MachineName      = $ServerHostname
                        ServerIP         = $ServerIP
                        Message          = $MsgText
                        EventData        = $XmlMap
                    })
                }
            } catch {
                if ($Ch -in @('System', 'Application', 'Security')) {
                    try {
                        $FallbackEvents = Get-EventLog -LogName $Ch -Newest 25 -ErrorAction Stop
                        foreach ($Ev in $FallbackEvents) {
                            $TimeIso = if ($Ev.TimeGenerated) { $Ev.TimeGenerated.ToString("yyyy-MM-ddTHH:mm:ss.fffzzz") } else { (Get-Date).ToString("o") }
                            $TimeDisp = if ($Ev.TimeGenerated) { $Ev.TimeGenerated.ToString("dd MMM yyyy, hh:mm:ss tt") } else { (Get-Date).ToString("dd MMM yyyy, hh:mm:ss tt") }
                            $AllEvents.Add([PSCustomObject]@{
                                Id               = $Ev.EventID
                                RecordId         = $Ev.Index
                                LogName          = $Ch
                                ProviderName     = $Ev.Source
                                LevelDisplayName = [string]$Ev.EntryType
                                Level            = 4
                                TimeCreated      = $TimeIso
                                TimeDisplay      = $TimeDisp
                                MachineName      = $ServerHostname
                                ServerIP         = $ServerIP
                                Message          = $Ev.Message
                                EventData        = @{ UserName = $Ev.UserName }
                            })
                        }
                    } catch {}
                }
            }
        }

        Write-Host "[$(Get-Date -Format 'HH:mm:ss')] Sent $($AllEvents.Count) detailed logs to SYSMIN Portal." -ForegroundColor Green

        $Payload = @{
            status    = "connected"
            hostname  = $ServerHostname
            ip        = $ServerIP
            isAdmin   = $IsAdmin
            timestamp = (Get-Date).ToString("o")
            logs      = $AllEvents
        } | ConvertTo-Json -Depth 5 -Compress

        $Buffer = [System.Text.Encoding]::UTF8.GetBytes($Payload)
        $Response.ContentType = "application/json; charset=utf-8"
        $Response.ContentLength64 = $Buffer.Length
        $Response.OutputStream.Write($Buffer, 0, $Buffer.Length)
        $Response.Close()
    } else {
        $Response.StatusCode = 404
        $Response.Close()
    }
}
